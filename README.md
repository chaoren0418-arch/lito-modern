pwsh -NoProfile -File .\啟動現代客戶端.ps1
pwsh -NoProfile -File .\啟動現代客戶端.ps1 -Verify

pwsh -NoProfile -File .\啟動現代客戶端.ps1 -NativeLoginEndpoint 127.0.0.1:<隔離登入端口>
